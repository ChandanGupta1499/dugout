# Fan Map (India, team-heatmap) Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a "Fan Map" tab to matches that have both teams, showing an India-centered map with two team-colored heatmap layers built from user-submitted (plus seeded) fan locations.

**Architecture:** New Express routes + JSON-file storage (`server/src/fans.ts` + `server/data/{teams,localities,fans,fans.seed}.json`) mirroring the existing `matches.ts` pattern; a new Expo Router screen (`src/app/match/[matchId]/fan-map.tsx`) using `react-native-maps`' `MapView` + `Heatmap`, with a one-time profile-setup flow cached in AsyncStorage (mirroring `src/lib/guest.ts`).

**Tech Stack:** Express (existing), `react-native-maps` (new dependency, Expo SDK 57 config plugin), `AsyncStorage` (existing), Expo Router (existing).

**Spec:** `docs/superpowers/specs/2026-09-09-fan-map-design.md` — read it before starting; this plan implements it task-by-task.

**Note on testing approach:** This repo has no test framework installed anywhere (no jest/vitest, no `*.test.*` files, `server/package.json` has no `test` script). Per the spec's own testing plan ("unit-level checks (or manual curl)... manually run the app"), this plan verifies server logic with manual `curl`/`node -e` checks and verifies the client by running it in the dev client — it does not introduce a new test framework. Do not add one as part of this feature.

```typescript
import { readFileSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

import type { MatchRecord } from './matches.js';

export type Team = { slug: string; label: string };
export type Locality = { id: string; name: string; lat: number; lng: number };
export type FanRecord = {
  userId: string;
  teamSlug: string;
  localityId: string;
  updatedAt: string;
};

const dataDir = join(dirname(fileURLToPath(import.meta.url)), '../data');
const teamsPath = join(dataDir, 'teams.json');
const localitiesPath = join(dataDir, 'localities.json');
const fansPath = join(dataDir, 'fans.json');
const fansSeedPath = join(dataDir, 'fans.seed.json');

let teamsCache: Team[] | null = null;
let localitiesCache: Locality[] | null = null;
let seedFansCache: FanRecord[] | null = null;

export function listTeams(): Team[] {
  if (!teamsCache) {
    teamsCache = JSON.parse(readFileSync(teamsPath, 'utf8')) as Team[];
  }
  return teamsCache;
}

export function listLocalities(): Locality[] {
  if (!localitiesCache) {
    localitiesCache = JSON.parse(readFileSync(localitiesPath, 'utf8')) as Locality[];
  }
  return localitiesCache;
}

export function searchLocalities(query: string): Locality[] {
  const q = query.trim().toLowerCase();
  const all = listLocalities();
  if (!q) {
    return all.slice(0, 20);
  }
  return all.filter((loc) => loc.name.toLowerCase().includes(q)).slice(0, 20);
}

function readRealFans(): FanRecord[] {
  // Read fresh every call (unlike teams/localities): this file is mutated by
  // POST /fans/profile and must never be served stale from a module cache.
  return JSON.parse(readFileSync(fansPath, 'utf8')) as FanRecord[];
}

function readSeedFans(): FanRecord[] {
  if (!seedFansCache) {
    seedFansCache = JSON.parse(readFileSync(fansSeedPath, 'utf8')) as FanRecord[];
  }
  return seedFansCache;
}

export function getFanProfile(userId: string): FanRecord | undefined {
  return readRealFans().find((fan) => fan.userId === userId);
}

export type UpsertFanInput = { userId: string; teamSlug: string; localityId: string };

export function upsertFanProfile(input: UpsertFanInput): FanRecord {
  const teamExists = listTeams().some((team) => team.slug === input.teamSlug);
  if (!teamExists) {
    throw new Error(`Unknown team: ${input.teamSlug}`);
  }
  const localityExists = listLocalities().some((loc) => loc.id === input.localityId);
  if (!localityExists) {
    throw new Error(`Unknown locality: ${input.localityId}`);
  }

  // Synchronous read-modify-write, no `await` in between: Node's single
  // event loop guarantees no other request can interleave here.
  const fans = readRealFans();
  const record: FanRecord = {
    userId: input.userId,
    teamSlug: input.teamSlug,
    localityId: input.localityId,
    updatedAt: new Date().toISOString(),
  };
  const index = fans.findIndex((fan) => fan.userId === input.userId);
  if (index === -1) {
    fans.push(record);
  } else {
    fans[index] = record;
  }
  writeFileSync(fansPath, JSON.stringify(fans, null, 2));
  return record;
}

export type FanMapPoint = { lat: number; lng: number; weight: number };
export type FanMapResponse = { teamA: FanMapPoint[]; teamB: FanMapPoint[] };

export function getFanMap(match: MatchRecord): FanMapResponse {
  if (!match.teamA || !match.teamB) {
    throw new Error(`Match ${match.id} does not have two teams`);
  }

  const localityById = new Map(listLocalities().map((loc) => [loc.id, loc]));
  const allFans = [...readRealFans(), ...readSeedFans()];

  function pointsForTeam(teamSlug: string): FanMapPoint[] {
    const counts = new Map<string, number>();
    for (const fan of allFans) {
      if (fan.teamSlug !== teamSlug) continue;
      if (!localityById.has(fan.localityId)) {
        console.warn(
          `Skipping fan record with unknown localityId "${fan.localityId}" (userId: ${fan.userId})`,
        );
        continue;
      }
      counts.set(fan.localityId, (counts.get(fan.localityId) ?? 0) + 1);
    }
    return [...counts.entries()].map(([localityId, weight]) => {
      const loc = localityById.get(localityId)!;
      return { lat: loc.lat, lng: loc.lng, weight };
    });
  }

  return {
    teamA: pointsForTeam(match.teamA.slug),
    teamB: pointsForTeam(match.teamB.slug),
  };
}
```

### Task 3: Seed data generator

**Files:**
- Create: `server/scripts/generate-fan-seed.ts`
- Modify: `server/data/fans.seed.json` (generated output)
- Modify: `server/package.json:scripts`

Real-world flavor: Argentina/Brazil fandom is famously concentrated in West Bengal and Kerala in India; England/European club fandom skews toward the larger metros. The generator encodes this as a rough per-team locality weighting rather than pure uniform randomness, using a fixed seed so the output is reproducible.

- [ ] **Step 1: Write `server/scripts/generate-fan-seed.ts`**

```typescript
import { writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';

import localities from '../data/localities.json' with { type: 'json' };
import teams from '../data/teams.json' with { type: 'json' };

// Simple deterministic PRNG (mulberry32) so regenerating this file is reproducible.
function mulberry32(seed: number) {
  return function random() {
    seed |= 0;
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const random = mulberry32(20260909);

// Higher weight = more likely to be picked for a given team. Base weight 1 for
// every locality; bumped for realistic regional fandom skews.
const KOLKATA_KERALA_IDS = new Set([
  'salt-lake-kolkata',
  'park-street-kolkata',
  'panampilly-nagar-kochi',
  'vazhuthacaud-thiruvananthapuram',
]);
const METRO_IDS = new Set([
  'worli-mumbai',
  'bandra-mumbai',
  'connaught-place-delhi',
  'koramangala-bengaluru',
  'indiranagar-bengaluru',
  'banjara-hills-hyderabad',
  'anna-nagar-chennai',
]);

function weightFor(teamSlug: string, localityId: string): number {
  if (teamSlug === 'argentina' && KOLKATA_KERALA_IDS.has(localityId)) return 6;
  if (METRO_IDS.has(localityId)) return 3;
  return 1;
}

function weightedPick<T extends { id: string }>(
  items: T[],
  weightOf: (item: T) => number,
): T {
  const total = items.reduce((sum, item) => sum + weightOf(item), 0);
  let roll = random() * total;
  for (const item of items) {
    roll -= weightOf(item);
    if (roll <= 0) return item;
  }
  return items[items.length - 1];
}

const RECORDS_PER_TEAM = 60;
const seedFans: { userId: string; teamSlug: string; localityId: string; updatedAt: string }[] = [];
let counter = 0;

for (const team of teams) {
  for (let i = 0; i < RECORDS_PER_TEAM; i++) {
    const locality = weightedPick(localities, (loc) => weightFor(team.slug, loc.id));
    seedFans.push({
      userId: `seed-${counter++}`,
      teamSlug: team.slug,
      localityId: locality.id,
      updatedAt: '2026-09-09T00:00:00.000Z',
    });
  }
}

const outPath = join(dirname(fileURLToPath(import.meta.url)), '../data/fans.seed.json');
writeFileSync(outPath, JSON.stringify(seedFans, null, 2));
console.log(`Wrote ${seedFans.length} seed fan records to ${outPath}`);
```

```typescript
app.post('/fans/profile', (req, res) => {
  const { userId, teamSlug, localityId } = req.body ?? {};

  if (typeof userId !== 'string' || !userId.trim()) {
    res.status(400).json({ error: 'userId is required' });
    return;
  }
  if (typeof teamSlug !== 'string' || !teamSlug.trim()) {
    res.status(400).json({ error: 'teamSlug is required' });
    return;
  }
  if (typeof localityId !== 'string' || !localityId.trim()) {
    res.status(400).json({ error: 'localityId is required' });
    return;
  }

  try {
    const profile = upsertFanProfile({
      userId: userId.trim(),
      teamSlug: teamSlug.trim(),
      localityId: localityId.trim(),
    });
    res.json(profile);
  } catch (error) {
    const message = error instanceof Error ? error.message : '';
    if (message.startsWith('Unknown team') || message.startsWith('Unknown locality')) {
      res.status(400).json({ error: message });
      return;
    }
    console.error('POST /fans/profile failed', error);
    res.status(500).json({ error: 'Failed to save fan profile' });
  }
});

app.get('/matches/:matchId/fan-map', (req, res) => {
  const match = getMatch(req.params.matchId);
  if (!match) {
    res.status(404).json({ error: 'Match not found' });
    return;
  }
  if (!match.teamA || !match.teamB) {
    res.status(404).json({ error: 'Fan map is not available for this match' });
    return;
  }
  res.json(getFanMap(match));
});
```

```typescript
import AsyncStorage from '@react-native-async-storage/async-storage';

const FAN_PROFILE_KEY = 'dugout.fanProfile';

export type CachedFanProfile = {
  teamSlug: string;
  localityId: string;
};

export async function loadCachedFanProfile(): Promise<CachedFanProfile | null> {
  const raw = await AsyncStorage.getItem(FAN_PROFILE_KEY);
  if (!raw) {
    return null;
  }
  try {
    const parsed = JSON.parse(raw) as CachedFanProfile;
    if (parsed?.teamSlug && parsed?.localityId) {
      return parsed;
    }
  } catch {
    // fall through and treat as absent
  }
  return null;
}

export async function saveCachedFanProfile(profile: CachedFanProfile): Promise<void> {
  await AsyncStorage.setItem(FAN_PROFILE_KEY, JSON.stringify(profile));
}
```

```typescript
import { useState } from 'react';
import { ActivityIndicator, Pressable, StyleSheet, Text, TextInput, View } from 'react-native';

import { saveFanProfile, searchLocalities, type Locality, type Team } from '@/lib/api';
import { saveCachedFanProfile } from '@/lib/fan-profile';

type Props = {
  userId: string;
  teams: Team[];
  onComplete: (profile: { teamSlug: string; localityId: string }) => void;
};

export function FanProfileSetup({ userId, teams, onComplete }: Props) {
  const [selectedTeam, setSelectedTeam] = useState<Team | null>(null);
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<Locality[]>([]);
  const [selectedLocality, setSelectedLocality] = useState<Locality | null>(null);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState<string | null>(null);

    setQuery(text);
    setSelectedLocality(null);
    if (!text.trim()) {
      setResults([]);
      return;
    }
    try {
      const next = await searchLocalities(text);
      setResults(next);
    } catch {
      setResults([]);
    }

    if (!selectedTeam || !selectedLocality || saving) {
      return;

    setError(null);

      await saveCachedFanProfile({
        teamSlug: selectedTeam.slug,
        localityId: selectedLocality.id,
      });
      onComplete({ teamSlug: selectedTeam.slug, localityId: selectedLocality.id });
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to save profile');
    } finally {
      setSaving(false);
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Which team do you support?</Text>
      <View style={styles.teamRow}>
        {teams.map((team) => (
          <Pressable
            key={team.slug}
            style={[styles.teamChip, selectedTeam?.slug === team.slug && styles.teamChipSelected]}

                styles.teamChipLabel,

        <Pressable
          key={loc.id}
          style={[styles.localityRow, selectedLocality?.id === loc.id && styles.localityRowSelected]}
          onPress={() => setSelectedLocality(loc)}>
          <Text>{loc.name}</Text>
        </Pressable>
      ))}

      {error ? <Text style={styles.error}>{error}</Text> : null}

      <Pressable
        style={[styles.confirmButton, (!selectedTeam || !selectedLocality) && styles.confirmButtonDisabled]}
        disabled={!selectedTeam || !selectedLocality || saving}
        onPress={() => void confirm()}>
        {saving ? <ActivityIndicator color="#fff" /> : <Text style={styles.confirmButtonLabel}>Save</Text>}
      </Pressable>
    </View>
  );
}
```

```typescript
import { useCallback, useEffect, useState } from 'react';
import { ActivityIndicator, Pressable, StyleSheet, Text, View } from 'react-native';
import { Stack, useLocalSearchParams } from 'expo-router';
import MapView, { Heatmap, PROVIDER_GOOGLE } from 'react-native-maps';

import { FanProfileSetup } from '@/components/fan-profile-setup';
import { fetchFanMap, fetchFanProfile, fetchMatch, fetchTeams, type FanMapResponse, type Team } from '@/lib/api';
import { loadCachedFanProfile, saveCachedFanProfile } from '@/lib/fan-profile';
import type { Match } from '@/lib/matches';
import { useGuest } from '@/providers/chat-provider';

const INDIA_REGION = {
  latitude: 22.5,
  longitude: 80.0,
  latitudeDelta: 20,
  longitudeDelta: 20,
};

export default function FanMapScreen() {
  const { matchId } = useLocalSearchParams<{ matchId: string }>();
  const guest = useGuest();

  const [match, setMatch] = useState<Match | null>(null);
  const [hasProfile, setHasProfile] = useState<boolean | null>(null); // null = still resolving
  const [profileCheckError, setProfileCheckError] = useState<string | null>(null);
  const [teams, setTeams] = useState<Team[]>([]);
  const [fanMap, setFanMap] = useState<FanMapResponse | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [profileRetryCount, setProfileRetryCount] = useState(0);
  const [mapRetryCount, setMapRetryCount] = useState(0);

  // Resolve profile existence once: check the local cache first, and only
  // fall back to a network call if nothing is cached yet, so a user with a
  // server-side profile but a fresh cache never sees the setup screen flash.
  // `fetchFanProfile` (Task 6) returns `null` on a real 404 ("no profile
  // exists") and only *throws* on an actual failure (network/500) — those
  // two cases must not be conflated, or a transient network blip would
  // wrongly push a user with an existing profile into the setup flow.
  useEffect(() => {
    let cancelled = false;
    (async () => {
      const cached = await loadCachedFanProfile();
      if (cached) {
        if (!cancelled) setHasProfile(true);
        return;
      }
      try {
        const remote = await fetchFanProfile(guest.userId);
        if (cancelled) return;
        if (remote) {
          await saveCachedFanProfile({ teamSlug: remote.teamSlug, localityId: remote.localityId });
          setHasProfile(true);
        } else {
          setHasProfile(false); // confirmed 404: no profile exists yet
        }
      } catch (err) {
        if (!cancelled) {
          setProfileCheckError(err instanceof Error ? err.message : 'Failed to check fan profile');
        }
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [guest.userId, profileRetryCount]);

  useEffect(() => {
    if (hasProfile === false) {
      void fetchTeams().then(setTeams).catch(() => setTeams([]));
    }
  }, [hasProfile]);

  const loadMapData = useCallback(async () => {
    if (!matchId) {
      setError('Missing match id');
      return;
    }
    setError(null);
    setFanMap(null);
    try {
      const [nextMatch, nextFanMap] = await Promise.all([
        fetchMatch(matchId),
        fetchFanMap(matchId),
      ]);
      setMatch(nextMatch);
      setFanMap(nextFanMap);
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Failed to load fan map');
    }
  }, [matchId]);

  useEffect(() => {
    if (hasProfile === true) {
      void loadMapData();
    }
  }, [hasProfile, loadMapData, mapRetryCount]);

  if (profileCheckError) {
    return (
      <View style={styles.centered}>
        <Text style={styles.message}>Could not check your fan profile.</Text>
        <Text style={styles.detail}>{profileCheckError}</Text>
        <Pressable
          style={styles.retryButton}
          onPress={() => {
            setProfileCheckError(null);
            setProfileRetryCount((c) => c + 1);
          }}>
          <Text style={styles.retryButtonLabel}>Retry</Text>
        </Pressable>
      </View>
    );
  }

  if (hasProfile === null) {
    return (
      <View style={styles.centered}>
        <ActivityIndicator />
        <Text style={styles.message}>Loading fan map…</Text>
      </View>
    );
  }

  if (hasProfile === false) {
    return (
      <>
        <Stack.Screen options={{ title: 'Set up your fan profile' }} />
        <FanProfileSetup userId={guest.userId} teams={teams} onComplete={() => setHasProfile(true)} />
      </>
    );
  }

  if (error) {
    return (
      <View style={styles.centered}>
        <Text style={styles.message}>Could not load fan map.</Text>
        <Text style={styles.detail}>{error}</Text>
        <Pressable style={styles.retryButton} onPress={() => setMapRetryCount((c) => c + 1)}>
          <Text style={styles.retryButtonLabel}>Retry</Text>
        </Pressable>
      </View>
    );
  }

  if (!fanMap || !match) {
    return (
      <View style={styles.centered}>
        <ActivityIndicator />
        <Text style={styles.message}>Loading fan map…</Text>
      </View>
    );
  }

  return (
    <>
      <Stack.Screen options={{ title: `${match.title} · Fan Map` }} />
      <MapView style={styles.map} initialRegion={INDIA_REGION} provider={PROVIDER_GOOGLE}>
        {fanMap.teamA.length > 0 ? (
          <Heatmap
            points={fanMap.teamA.map((p) => ({ latitude: p.lat, longitude: p.lng, weight: p.weight }))}
            radius={40}
            opacity={0.7}
            gradient={{
              colors: ['#ffe0e0', '#ff4d4d', '#b30000'],
              startPoints: [0.1, 0.5, 1],
              colorMapSize: 256,
            }}
          />
        ) : null}
        {fanMap.teamB.length > 0 ? (
          <Heatmap
            points={fanMap.teamB.map((p) => ({ latitude: p.lat, longitude: p.lng, weight: p.weight }))}
            radius={40}
            opacity={0.7}
            gradient={{
              colors: ['#e0f0ff', '#4da6ff', '#0047b3'],
              startPoints: [0.1, 0.5, 1],
              colorMapSize: 256,
            }}
          />
        ) : null}
      </MapView>
    </>
  );
}

const styles = StyleSheet.create({
  map: { flex: 1 },
  centered: {
    flex: 1,
    alignItems: 'center',
    justifyContent: 'center',
    padding: 24,
    gap: 12,
    backgroundColor: '#fff',
  },
  message: { fontSize: 16, textAlign: 'center' },
  detail: { fontSize: 13, color: '#666', textAlign: 'center' },
  retryButton: {
    marginTop: 8,
    backgroundColor: '#111',
    paddingHorizontal: 16,
    paddingVertical: 10,
    borderRadius: 8,
  },
  retryButtonLabel: { color: '#fff', fontWeight: '600' },
});
```
